import com.fafa.infrastructure.persistence.mapper.UserTagMapper;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Repository;
import java.util.List;
import java.util.Optional;
import java.util.stream.Collectors;

@Repository
@RequiredArgsConstructor
public class UserTagRepositoryImpl implements UserTagRepository {
    
    private final UserTagMapper userTagMapper;
    
    @Override
    public UserTag save(UserTag userTag) {
        UserTagPO po = toPO(userTag);
        if (po.getId() == null) {
            userTagMapper.insert(po);
        } else {
            userTagMapper.updateById(po);
        }
        return toDomain(po);
    }
    
    @Override
    public Optional<UserTag> findById(UserTagId id) {
        UserTagPO po = userTagMapper.selectById(id.getValue());
        return Optional.ofNullable(po).map(this::toDomain);
    }
    
    @Override
    public List<UserTag> findByUserId(Long userId) {
        return userTagMapper.selectByUserId(userId).stream()
                .map(this::toDomain)
                .collect(Collectors.toList());
    }
    
    @Override
    public List<UserTag> findByUserIdOrderByUsageCountDesc(Long userId, int limit) {
        return userTagMapper.selectByUserIdOrderByUsage(userId, limit).stream()
                .map(this::toDomain)
                .collect(Collectors.toList());
    }
    
    @Override
    public Optional<UserTag> findByUserIdAndTagName(Long userId, String tagName) {
        UserTagPO po = userTagMapper.selectByUserIdAndTagName(userId, tagName);
        return Optional.ofNullable(po).map(this::toDomain);
    }
    
    @Override
    public void deleteById(UserTagId id) {
        userTagMapper.deleteById(id.getValue());
    }
    
    @Override
    public void deleteByIds(List<UserTagId> ids) {
        List<Long> idValues = ids.stream().map(UserTagId::getValue).collect(Collectors.toList());
        userTagMapper.deleteBatchIds(idValues);
    }
    
    @Override
    public int countByUserId(Long userId) {
        return userTagMapper.countByUserId(userId);
    }

    private UserTag toDomain(UserTagPO po) { /* ... */ }

    log.info("上传宠物三视图: petId={}, userId={}", petId, userId);

    Pet pet = petRepository.findById(new PetId(petId))

    String topUrl = ossService.uploadFile(topView, "pets/profiles");
    
    // 3. 更新宠物档案
    pet.updateProfilePhotos(frontUrl, sideUrl, topUrl);
    petRepository.save(pet);
    
    // 4. 发送 MQ 消息给 Python 服务处理向量化

        PetProfileMessage message = PetProfileMessage.builder()
                .petId(petId)
                .userId(userId)
                .frontViewUrl(frontUrl)
                .sideViewUrl(sideUrl)
                .topViewUrl(topUrl)
                .build();

    } catch (Exception e) {
        log.error("发送宠物档案消息失败，petId={}", petId, e);
    }
    
    log.info("宠物三视图上传成功: petId={}", petId);

/**
 * 上传照片/视频
 * 
 * @param userId 用户ID
 * @param petId 宠物ID (可为 null, AI自动识别)
 * @param file 文件
 * @param takenAt 拍摄时间
 * @param description 描述
 * @param tags 用户标签
 */
@Transactional(rollbackFor = Exception.class)
public Long uploadPhoto(Long userId, Long petId, MultipartFile file, 
                       LocalDateTime takenAt, String description, List<String> tags) {
    
    // 1. 文件类型检测
    String contentType = file.getContentType();
    String mediaType = "image";
    
    if (contentType != null) {
        if (contentType.startsWith("video/")) {
            mediaType = "video";
            // 视频大小限制 50MB
            if (file.getSize() > 50 * 1024 * 1024) {
                throw new BusinessException("视频大小不能超过 50MB");
            }
        } else if (contentType.startsWith("image/")) {
            // 图片大小限制 15MB
            if (file.getSize() > 15 * 1024 * 1024) {
                throw new BusinessException("图片大小不能超过 15MB");
            }
        } else {
            throw new BusinessException("只支持图片和视频文件");
        }
    }
    
    // 2. 如果 petId 为空但提供了 tags, 尝试从 tags 中推断宠物
    if (petId == null && tags != null && !tags.isEmpty()) {
        // 从 tags 中查找是否有宠物名称
        petId = inferPetIdFromTags(userId, tags);
    }
    
    // 3. 上传文件到 OSS
    String originalUrl = ossService.uploadFile(file, mediaType.equals("video") ? "videos" : "photos");
    String thumbnailUrl = originalUrl;
    String videoCoverUrl = null;
    Integer duration = null;
    
    // 4. 如果是视频, 生成封面
    if ("video".equals(mediaType)) {
        videoCoverUrl = ossService.generateVideoThumbnail(originalUrl);
        thumbnailUrl = videoCoverUrl;
        // duration 由 Python 服务分析后回填
    }
    
    // 5. 读取图片元数据
    Integer width = null;
    Integer height = null;
    if ("image".equals(mediaType)) {
        try {
            BufferedImage image = ImageIO.read(file.getInputStream());
            if (image != null) {
                width = image.getWidth();
                height = image.getHeight();
            }
        } catch (IOException e) {
            log.warn("读取图片尺寸失败: {}", e.getMessage());
        }
    }
    
    // 6. 创建照片/视频记录
    Photo photo = Photo.create(petId, userId, originalUrl, thumbnailUrl, 
                               takenAt != null ? takenAt : LocalDateTime.now(), 
                               description, mediaType);
    photo.updateMetadata(width, height, file.getSize(), originalUrl);
    
    if ("video".equals(mediaType)) {
        photo.updateVideoInfo(duration, videoCoverUrl);
    }
    
    // 7. 设置用户标签
    if (tags != null && !tags.isEmpty()) {
        photo.setTags(tags);
        // 更新标签使用计数
        updateTagUsageCount(userId, tags);
    }
    
    Photo savedPhoto = photoRepository.save(photo);
    
    // 8. 发送 MQ 消息触发 AI 分析
    try {
        PhotoAnalysisMessage message = PhotoAnalysisMessage.builder()
                .photoId(savedPhoto.getPhotoId().getValue())
                .petId(petId)
                .userId(userId)
                .url(originalUrl)
                .thumbnailUrl(thumbnailUrl)
                .takenAt(savedPhoto.getTakenAt().toString())
                .mediaType(mediaType)
                .tags(tags)
                .isProfilePhoto(false)
                .build();
        
        mqProducerService.sendPhotoAnalysisMessage(message);
    } catch (Exception e) {
        log.error("发送照片分析消息失败，photoId={}", savedPhoto.getPhotoId().getValue(), e);
    }
    
    log.info("媒体上传成功: photoId=, mediaType={}, petId={}, userId={}", 
            savedPhoto.getPhotoId().getValue(), mediaType, petId, userId);
    
    return savedPhoto.getPhotoId().getValue();
}

/**
 * 从标签中推断宠物ID
 */
private Long inferPetIdFromTags(Long userId, List<String> tags) {
    // 查询用户的所有宠物
    List<Pet> pets = petRepository.findByUserId(userId);
    
    // 查找标签中是否包含宠物名称
    for (String tag : tags) {
        for (Pet pet : pets) {
            if (pet.getName().equalsIgnoreCase(tag.trim())) {
                return pet.getId().getValue();
            }
        }
    }
    
    return null;
}

/**
 * 更新标签使用计数
 */
private void updateTagUsageCount(Long userId, List<String> tags) {
    for (String tagName : tags) {
        Optional<UserTag> tagOpt = userTagRepository.findByUserIdAndTagName(userId, tagName);
        if (tagOpt.isPresent()) {
            UserTag tag = tagOpt.get();
            tag.incrementUsage();
            userTagRepository.save(tag);
        } else {
            // 自动创建新标签
            UserTag newTag = UserTag.create(userId, tagName, "other");
            newTag.incrementUsage();
            userTagRepository.save(newTag);
        }
    }
}

 * 批量上传照片/视频
 */
@Transactional(rollbackFor = Exception.class)
public List<Long> batchUploadPhotos(Long userId, Long petId, List<MultipartFile> files,

        photoIds.add(photoId);
    }
    
    log.info("批量上传完成: userId={}, count={}", userId, photoIds.size());

}

package com.fafa.application.service;

import com.fafa.common.exception.BusinessException;
import com.fafa.domain.model.tag.UserTag;
import com.fafa.domain.model.tag.UserTagId;
import com.fafa.domain.repository.UserTagRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.util.List;
import java.util.stream.Collectors;

@Slf4j
@Service
@RequiredArgsConstructor
public class UserTagApplicationService {

    private final UserTagRepository userTagRepository;

    /**
     * 创建标签
     */
    @Transactional(rollbackFor = Exception.class)
    public Long createTag(Long userId, String tagName, String category) {
        log.info("创建标签: userId={}, tagName={}, category={}", userId, tagName, category);
        
        // 检查是否已存在
        if (userTagRepository.findByUserIdAndTagName(userId, tagName).isPresent()) {
            throw new BusinessException("标签已存在");
        }
        
        // 检查标签数量限制
        int count = userTagRepository.countByUserId(userId);
        if (count >= 100) {
            throw new BusinessException("标签数量已达上限(100个)");
        }
        
        UserTag tag = UserTag.create(userId, tagName, category);
        UserTag saved = userTagRepository.save(tag);
        
        return saved.getId().getValue();
    }

    /**
     * 查询用户的所有标签
     */
    public List<UserTag> listUserTags(Long userId) {
        return userTagRepository.findByUserId(userId);
    }

    /**
     * 获取标签建议(按使用频率排序)
     */
    public List<UserTag> getTagSuggestions(Long userId, int limit) {
        return userTagRepository.findByUserIdOrderByUsageCountDesc(userId, limit);
    }

    /**
     * 删除标签
     */
    @Transactional(rollbackFor = Exception.class)
    public void deleteTag(Long userId, Long tagId) {
        UserTag tag = userTagRepository.findById(UserTagId.of(tagId))
                .orElseThrow(() -> new BusinessException("标签不存在"));
        
        if (!tag.getUserId().equals(userId)) {
            throw new BusinessException("无权删除该标签");
        }
        
        userTagRepository.deleteById(UserTagId.of(tagId));
        log.info("删除标签成功: tagId={}, userId={}", tagId, userId);
    }

    /**
     * 批量删除标签
     */
    @Transactional(rollbackFor = Exception.class)
    public void batchDeleteTags(Long userId, List<Long> tagIds) {
        // 验证所有标签归属
        for (Long tagId : tagIds) {
            UserTag tag = userTagRepository.findById(UserTagId.of(tagId))
                    .orElseThrow(() -> new BusinessException("标签不存在: " + tagId));
            
            if (!tag.getUserId().equals(userId)) {
                throw new BusinessException("无权删除标签: " + tagId);
            }
        }
        
        List<UserTagId> ids = tagIds.stream()
                .map(UserTagId::of)
                .collect(Collectors.toList());
        
        userTagRepository.deleteByIds(ids);
        log.info("批量删除标签成功: userId={}, count={}", userId, tagIds.size());
    }
}

        @Parameter(description = "宠物 ID", required = true)
        @PathVariable Long id,
        @Parameter(description = "正面照", required = true)
        @RequestParam("frontView") MultipartFile frontView,
        @Parameter(description = "侧面照", required = true)

    validateImageFile(topView, "俯视照");

    return Result.success();
}

private void validateImageFile(MultipartFile file, String name) {
    if (file.isEmpty()) {
        throw new BusinessException(name + "不能为空");
    }
    
    String contentType = file.getContentType();
    if (contentType == null || !contentType.startsWith("image/")) {
        throw new BusinessException(name + "必须是图片文件");
    }
    
    if (file.getSize() > 10 * 1024 * 1024) {
        throw new BusinessException(name + "大小不能超过 10MB");
    }
}

"""
通义千问 API 客户端 - qwen3-vl-embedding 版本

import dashscope
from dashscope import MultiModalEmbedding
from typing import Union, List, Dict
from loguru import logger
from app.core.config import settings

dashscope.api_key = settings.DASHSCOPE_API_KEY


async def generate_multimodal_embedding(
    content: Union[str, Dict[str, str]],
    dimension: int = 1024
) -> List[float]:
    """
    生成多模态向量
    
    Args:
        content: 
            - 文本: "一只猫在阳台"
            - 图片: {"image": "https://..."}
            - 视频: {"video": "https://..."}
        dimension: 向量维度 (Matryoshka支持: 512, 768, 1024)
    
    Returns:
        向量数组
    """
    try:
        response = MultiModalEmbedding.call(
            model='qwen3-vl-embedding',
            input=content,
            dimension=dimension
        )
        
        if response.status_code == 200:
            embedding = response.output.embeddings[0].embedding
            logger.info(f"生成向量成功, 维度: {len(embedding)}")
            return embedding
        else:
            logger.error(f"生成向量失败: {response.code}, {response.message}")
            raise Exception(f"向量生成失败: {response.message}")
            
    except Exception as e:
        logger.error(f"调用 qwen3-vl-embedding 失败: {e}", exc_info=True)
        raise

"""
照片/视频分析服务 - qwen3-vl-embedding 版本
"""
import asyncio
from typing import List, Dict, Any, Optional
from loguru import logger
from app.core.config import settings
from app.core.database import get_async_session
from app.core.qdrant import qdrant_client
from app.core.dashscope_client import generate_multimodal_embedding
from app.repository.photo_repository import PhotoRepository
from app.service.vector_service import VectorService
from sqlalchemy import text


class PhotoAnalysisService:
    """照片/视频分析服务"""
    
    def __init__(self):
        self.photo_repo = PhotoRepository()
        self.vector_service = VectorService()
    
    async def analyze_media(
        self, 
        photo_id: int, 
        user_id: int,
        pet_id: Optional[int], 
        url: str,
        media_type: str,
        tags: Optional[List[str]] = None
    ):
        """
        分析照片/视频
        
        Args:
            photo_id: 照片ID
            user_id: 用户ID
            pet_id: 宠物ID (可为None)
            url: 媒体URL
            media_type: 媒体类型 "image" | "video"
            tags: 用户标签
        """
        try:
            logger.info(f"开始分析媒体: photoId={photo_id}, mediaType={media_type}")
            
            # 1. 如果 petId 为空,尝试自动识别宠物
            recognized_pet_id = pet_id
            recognition_confidence = None
            auto_recognized = False
            
            if pet_id is None:
                recognition_result = await self._recognize_pet(user_id, url, media_type)
                if recognition_result:
                    recognized_pet_id = recognition_result['pet_id']
                    recognition_confidence = recognition_result['confidence']
                    auto_recognized = True
                    logger.info(f"自动识别宠物成功: petId={recognized_pet_id}, confidence={recognition_confidence}")
            
            # 2. 生成向量
            embedding = await self._generate_embedding(url, media_type)
            
            if not embedding:
                logger.error(f"生成向量失败: photoId={photo_id}")
                return
            
            # 3. 存入 Qdrant
            embedding_id = f"{media_type}_{photo_id}"
            await self._save_to_qdrant(
                embedding_id=embedding_id,
                vector=embedding,
                payload={
                    'media_id': photo_id,
                    'user_id': user_id,
                    'pet_id': recognized_pet_id,
                    'media_type': media_type,
                    'url': url,
                    'tags': tags or [],
                    'auto_recognized': auto_recognized,
                    'recognition_confidence': recognition_confidence
                }
            )
            
            # 4. 更新 MySQL
            await self._update_photo_result(
                photo_id=photo_id,
                pet_id=recognized_pet_id,
                embedding_id=embedding_id,
                auto_recognized=auto_recognized,
                recognition_confidence=recognition_confidence
            )
            
            logger.info(f"媒体分析完成: photoId={photo_id}, embeddingId={embedding_id}")
            
        except Exception as e:
            logger.error(f"媒体分析失败: photoId={photo_id}, error={e}", exc_info=True)
            raise
    
    async def _recognize_pet(
        self, 
        user_id: int, 
        media_url: str, 
        media_type: str
    ) -> Optional[Dict[str, Any]]:
        """
        识别宠物 (图片到图片相似度搜索)
        
        Args:
            user_id: 用户ID
            media_url: 媒体URL
            media_type: 媒体类型
        
        Returns:
            {'pet_id': xx, 'pet_name': 'xx', 'confidence': 0.85} or None
        """
        try:
            # 1. 生成媒体向量
            content = {"image": media_url} if media_type == "image" else {"video": media_url}
            media_vector = await generate_multimodal_embedding(content, dimension=settings.QWEN3_VL_EMBEDDING_DIM)
            
            # 2. 在 pet_profiles collection 中搜索
            from qdrant_client.models import Filter, FieldCondition, MatchValue
            
            search_result = await qdrant_client.search(
                collection_name=settings.QDRANT_COLLECTION_PET_PROFILES,
                query_vector=media_vector,
                query_filter=Filter(
                    must=[
                        FieldCondition(
                            key="user_id",
                            match=MatchValue(value=user_id)
                        )
                    ]
                ),
                limit=1,
                score_threshold=settings.PET_RECOGNITION_THRESHOLD
            )
            
            if search_result and len(search_result) > 0:
                result = search_result[0]
                return {
                    'pet_id': result.payload['pet_id'],
                    'pet_name': result.payload['pet_name'],
                    'confidence': result.score
                }
            
            logger.info(f"未识别到宠物: userId={user_id}")
            return None
            
        except Exception as e:
            logger.error(f"宠物识别失败: {e}", exc_info=True)
            return None
    
    async def _generate_embedding(self, url: str, media_type: str) -> Optional[List[float]]:
        """
        生成向量
        
        Args:
            url: 媒体URL
            media_type: 媒体类型
        
        Returns:
            向量数组
        """
        try:
            if media_type == "image":
                content = {"image": url}
            elif media_type == "video":
                content = {"video": url}
            else:
                raise ValueError(f"不支持的媒体类型: {media_type}")
            
            embedding = await generate_multimodal_embedding(
                content=content,
                dimension=settings.QWEN3_VL_EMBEDDING_DIM
            )
            
            return embedding
            
        except Exception as e:
            logger.error(f"生成向量失败: {e}", exc_info=True)
            return None
    
    async def _save_to_qdrant(self, embedding_id: str, vector: List[float], payload: Dict[str, Any]):
        """保存向量到 Qdrant"""
        try:
            from qdrant_client.models import PointStruct
            
            await qdrant_client.upsert(
                collection_name=settings.QDRANT_COLLECTION_MEDIA,
                points=[
                    PointStruct(
                        id=embedding_id,
                        vector=vector,
                        payload=payload
                    )
                ]
            )
            
            logger.info(f"向量保存成功: embeddingId={embedding_id}")
            
        except Exception as e:
            logger.error(f"保存向量失败: {e}", exc_info=True)
            raise
    
    async def _update_photo_result(
        self,
        photo_id: int,
        pet_id: Optional[int],
        embedding_id: str,
        auto_recognized: bool,
        recognition_confidence: Optional[float]

                sql = text("""
                    UPDATE photo 
                    SET pet_id = :pet_id,
                        embedding_id = :embedding_id,
                        auto_recognized = :auto_recognized,
                        recognition_confidence = :recognition_confidence,
                        updated_at = NOW()
                    WHERE id = :photo_id
                """)
                
                await session.execute(sql, {
                    'pet_id': pet_id,
                    'embedding_id': embedding_id,
                    'auto_recognized': 1 if auto_recognized else 0,
                    'recognition_confidence': recognition_confidence,
                    'photo_id': photo_id
                })
                
                await session.commit()
                
            logger.info(f"更新照片结果成功: photoId={photo_id}")
            
        except Exception as e:
            logger.error(f"更新照片结果失败: {e}", exc_info=True)
            raise
